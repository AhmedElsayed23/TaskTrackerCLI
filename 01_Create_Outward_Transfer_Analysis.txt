            IF TEMP.AMT EQ "" THEN
                CALL CALCULATE.CHARGE(MAIN.CUSTOMER,Y.DEAL.AMOUNT, Y.DEAL.CCY, "11", Y.COMMISSION.RATE ,ORG.ACC.CUR.SRC , Y.COMM.CHARGE ,NEW.SEND.COMM , Y.CUST.CONDITION, CH.TOT.CHARGE.LCCY, CH.TOT.CHARGE.FCCY)
                IF Y.COMM.CHARGE EQ "EGP" THEN
                    TEMP.AMT = NEW.SEND.COMM<4,1>
                END
                ELSE
                    TEMP.AMT = NEW.SEND.COMM<5,1>
                END
            END
        END
        ELSE
            TEMP.CCY = TEMP.AMT[1,3]
            NEW.TEMP.AMT = TEMP.AMT[4,20]
            IF Y.COMM.CHARGE NE TEMP.CCY THEN
                CCY1 = TEMP.CCY
                AMT1 = NEW.TEMP.AMT
                CCY2 = Y.COMM.CHARGE
                AMT2 = ""
                EXCH.RATE = Y.COMM.RATE
                CCY.MKT = "11"
                BASE.CCY = ""
                CALL EXCHRATE(CCY.MKT, CCY1, AMT1, CCY2, AMT2, BASE.CCY, EXCH.RATE, "", LOCAL.AMOUNT, "")
                NEW.TEMP.AMT = AMT2
                TEMP.AMT = NEW.TEMP.AMT
            END
            ELSE
                TEMP.AMT = NEW.TEMP.AMT
            END
            CUST.PERCENT = TRIM(FIELD(SEND.COMMISSIONS.WAIVER,FM,TEMP.X))
            IF CUST.PERCENT NE 100 THEN

                IF CUST.PERCENT EQ "" THEN        ;* Note: NOT 0
                    CUST.PERCENT = 1    ;* 100% of charge to be borne
                END ELSE
                    CUST.PERCENT = 100 - CUST.PERCENT
                    CUST.PERCENT = CUST.PERCENT/100         ;* multiplication factor
                END
                TEMP.AMT = TEMP.AMT * CUST.PERCENT
                CAL = "1"
                CALL EB.ROUND.AMOUNT(Y.COMM.CHARGE,TEMP.AMT,CAL,"")
            END
            ELSE
                TEMP.AMT = "0"
            END
        END
        LOCATE "COMMISSION.AMT" IN OUT.TAG.VALUE.ARRAY SETTING POS THEN
            COMM.AMT.POS = POS
        END
        IF TEMP.AMT EQ "" THEN

        END
        ELSE
            IF NOT(ALPHA(TEMP.AMT[1,1])) THEN
                OUT.VALUES.ARRAY<POS>["_",TEMP.X,1] = Y.COMM.CHARGE:" ":TEMP.AMT
            END
            ELSE
                OUT.VALUES.ARRAY<POS>["_",TEMP.X,1] = TEMP.AMT
                TEMP.AMT = TRIM(TEMP.AMT[4,20])
            END
        END
        CONC.COMM.AMTS<TEMP.X> = TEMP.AMT

        TEMP.CON.FIELD = TRIM(FIELD(NEW.CONCESSION,"_",TEMP.X))
        IF TEMP.X NE "1" THEN
            CONC.FLAG := "_"
        END
        IF TEMP.CON.FIELD EQ "N" OR TEMP.CON.FIELD EQ "" THEN
            CONC.FLAG := "N"
        END
        ELSE
            CONC.FLAG := TEMP.CON.FIELD
        END

    NEXT TEMP.X

    IF CONC.FLAG NE '' THEN
        GOSUB CHECK.CONCESSION
    END

    RETURN
*****************************************************************************
CHECK.CONCESSION:
******************
    CONC.CUST.RATE<1> =  Y.COMM.RATE
    CONC.CUST.RATE<2> = Y.USD.RATE
    CONC.FLAG = ""
    CALL CIB.CALCULATE.CONCESSION(ORG.ACCOUNT,CHNL.CODE,Y.DEAL.AMOUNT,Y.DEAL.CCY,CONC.COMM.TYPES,CONC.COMM.AMTS,Y.COMM.CHARGE,CONC.COMM.WAIVERS,CONC.CUST.RATE,CONC.FLAG)
    CONCESSION.FLAG = "N"
    COUNTER.CONCESSION = DCOUNT(CONC.FLAG,FM)
    CONCESSION.FLAG = CONC.FLAG
    FOR TEMP.X = 1 TO COUNTER.CONCESSION
        TEMP.VALUE = FIELD(CONC.FLAG,FM,TEMP.X)
        IF TEMP.VALUE EQ "Y" THEN
*CONCESSION.FLAG = "Y"
            TEMP.AMT = CONC.COMM.AMTS<TEMP.X>
            OUT.VALUES.ARRAY<COMM.AMT.POS>["_",TEMP.X,1] = Y.COMM.CHARGE:" ":TEMP.AMT
        END
    NEXT TEMP.X
    CHANGE FM TO "_" IN CONCESSION.FLAG

    RETURN

*****************************************************************************
PROGRAM.END:
************
    RETURN
END
